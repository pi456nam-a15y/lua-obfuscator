from flask import Flask from threading import Thread import discord from discord.ext import commands # 1. สร้าง Web Server สั้นๆ ด้วย Flask app = Flask('') @app.route('/') def home(): return "Bot is running 24/7!" def run_web(): app.run(host='0.0.0.0', port=8080) def keep_alive(): t = Thread(target=run_web) t.start() # 2. โค้ดบอท Discord TOKEN = 'วาง_TOKEN_ของคุณตรงนี้' class MyBot(commands.Bot): def __init__(self): intents = discord.Intents.default() intents.message_content = True super().__init__(command_prefix="!", intents=intents) bot = MyBot() @bot.event async def on_ready(): print(f'บอท {bot.user} ออนไลน์แล้ว!') # เรียกให้ Web Server ทำงานก่อนรันบอท keep_alive() bot.run(TOKEN)
